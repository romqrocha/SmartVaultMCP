class Solution {
public:
    vector<vector<int>> mergeSimilarItems(vector<vector<int>>& items1, vector<vector<int>>& items2) {

        map<int,int> mp; // create map to add the 2 - 2D vectors to

        // We create sz1 and sz2 to avoid invoking the size() method multiple times in the
        // the 2 for loops that follow below.
        size_t sz1 = items1.size();
        size_t sz2 = items2.size();

        // Add 2D vector items1 to the map
        for(int i = 0; i < sz1;++i) {
            mp.insert(std::pair<int,int>(items1[i][0],items1[i][1]));
        }

        // Add 2D vector items2 to the map
        for(int i = 0; i < sz2;++i) {
            if(mp.find(items2[i][0]) != mp.end())
            {
               mp[items2[i][0]] += items2[i][1]; // if item was in items1
            }
            else
            {
                mp[items2[i][0]] = items2[i][1]; // if item was not in items1
            }
        }
       
        vector<vector<int>> result; // 2D vector we will store result into

        for(auto item : mp) { // iterate through items in map
                vector<int> v; // create temporary 1D vector to push onto result

                // put value and weight into the 1D vector
                v.push_back(item.first);
                v.push_back(item.second);

                result.push_back(v); // push the 1D vector onto the 2D vector
        }

        // We are done. Now simply return result 2D vector
        return result;
    }
};
